    Snippets Esenciales

    Servidor HTTP Mínimo

    import http from 'node:http';

http.createServer((req, res) => {
  res.writeHead(200, { 'Content-Type': 'text/plain' });
  res.end('Hola Mundo');
}).listen(3000);

    Servidor HTTPS Seguro

    import https from 'node:https';
import fs from 'node:fs';

const options = {
  key: fs.readFileSync('server-key.pem'),
  cert: fs.readFileSync('server-cert.pem'),
};

https.createServer(options, (req, res) => {
  res.writeHead(200);
  res.end('HTTPS seguro');
}).listen(443);

    Leer Body POST

    let body = '';
req.on('data', (chunk) => { body += chunk; });
req.on('end', () => {
  const data = JSON.parse(body);
  // procesar...
});

    Hacer Solicitud HTTP

    import http from 'node:http';

http.get('http://api.ejemplo.com/datos', (res) => {
  let data = '';
  res.on('data', (chunk) => { data += chunk; });
  res.on('end', () => {
    console.log(JSON.parse(data));
  });
}).on('error', (err) => console.error(err));

    Headers HTTP Comunes

    Header Ejemplo Propósito
    Content-Type application/json Tipo de contenido
    Content-Length 1024 Tamaño en bytes
    Cache-Control max-age=3600 Política de caché
    Accept-Encoding gzip, deflate Compresión soportada
    Authorization Bearer token Autenticación
    Set-Cookie sesion=abc Establece cookie
    Access-Control-Allow-Origin * CORS
    X-Custom-Header valor Header personalizado

    Códigos HTTP Principales

    Código Significado Caso de uso
    200 OK Solicitud exitosa
    201 Created Recurso creado
    204 No Content Éxito sin cuerpo
    301 Moved Permanently Redirección permanente
    302 Found Redirección temporal
    304 Not Modified Caché válido
    400 Bad Request Solicitud inválida
    401 Unauthorized Autenticación requerida
    403 Forbidden Acceso denegado
    404 Not Found Recurso no existe
    429 Too Many Requests Rate limit excedido
    500 Internal Server Error Error del servidor
    503 Service Unavailable Servicio no disponible

    Métodos HTTP

    Método Propósito Idempotente Con Body
    GET Obtener recurso Sí No
    POST Crear recurso No Sí
    PUT Actualizar (reemplazar) Sí Sí
    PATCH Actualizar (parcial) No Sí
    DELETE Eliminar recurso Sí No
    HEAD Como GET, sin body Sí No
    OPTIONS Información de métodos Sí No

    Propiedades Clave de Request (IncomingMessage)

    interface IncomingMessage {
  method: string;                    // GET, POST, PUT, DELETE, etc.
  url: string;                       // /ruta?param=valor
  httpVersion: string;               // '1.0', '1.1'
  headers: { [key: string]: string }; // Headers de la solicitud
  socket: Socket;                    // Socket TCP subyacente

  // Métodos
  on(event: string, callback: Function): void;
  setEncoding(encoding: string): void;
}

    Propiedades Clave de Response (ServerResponse)

    interface ServerResponse {
  statusCode: number;                // Código HTTP (200, 404, etc.)
  statusMessage: string;             // Mensaje de estado
  headersSent: boolean;              // ¿Se han enviado headers?
  finished: boolean;                 // ¿Se ha cerrado la respuesta?

  // Métodos
  writeHead(statusCode, headers?): void;
  setHeader(name: string, value: string): void;
  write(chunk: string | Buffer): boolean; // Retorna backpressure
  end(chunk?: string | Buffer): void;
  redirect(statusCode?: number, location: string): void;
}

    Manejo de Errores Típicos

    // Error: Puerto ya en uso
server.on('error', (err: NodeJS.ErrnoException) => {
  if (err.code === 'EADDRINUSE') {
    console.error('Puerto 3000 ya está en uso');
    process.exit(1);
  }
});

// Error: Headers ya enviados
try {
  res.writeHead(200);
  res.writeHead(500); // ❌ Error: ya se enviaron headers
} catch (err) {
  // El error se dispara implícitamente
}

// Error: Body muy grande
req.on('data', (chunk) => {
  if (totalSize > MAX_SIZE) {
    req.destroy(); // Detener solicitud
  }
});

// Timeout
server.requestTimeout = 30000; // 30 segundos

    Comparativa: http vs https

    // HTTP (sin encriptación)
import http from 'node:http';
http.createServer(callback).listen(80);

// HTTPS (con encriptación TLS)
import https from 'node:https';
import fs from 'node:fs';

const options = {
  key: fs.readFileSync('key.pem'),
  cert: fs.readFileSync('cert.pem'),
};
https.createServer(options, callback).listen(443);

// API idéntica, solo que https requiere certificados

    Keep-Alive y Persistencia de Conexiones

    // HTTP/1.1 (default): keep-alive activado
// Misma conexión TCP para múltiples solicitudes
// Cliente → GET /recurso1
// Servidor ← Respuesta 1
// Cliente → GET /recurso2 (misma conexión)
// Servidor ← Respuesta 2

// Configuración en Node.js
server.keepAliveTimeout = 65000;    // 65 segundos
server.headersTimeout = 66000;      // Timeout headers
server.requestTimeout = 300000;     // Timeout solicitud

    Seguridad: Headers Recomendados

    res.setHeader('X-Content-Type-Options', 'nosniff');
res.setHeader('X-Frame-Options', 'DENY');
res.setHeader('X-XSS-Protection', '1; mode=block');
res.setHeader('Strict-Transport-Security', 'max-age=31536000');
res.setHeader('Content-Security-Policy', "default-src 'self'");
res.setHeader('Referrer-Policy', 'strict-origin-when-cross-origin');

    Generar Certificado Autofirmado (Desarrollo)

    # Generar clave privada y certificado
openssl req -x509 -newkey rsa:2048 -nodes -sha256 \
  -subj '/CN=localhost' \
  -keyout server-key.pem \
  -out server-cert.pem \
  -days 365

# Verificar certificado
openssl x509 -in server-cert.pem -text -noout

    Debugging y Testing

    # Ver headers HTTP
curl -v http://localhost:3000

# POST con datos JSON
curl -X POST http://localhost:3000/api \
  -H "Content-Type: application/json" \
  -d '{"clave":"valor"}'

# Ignorar certificado autofirmado
curl -k https://localhost:443

# Ver procesos usando puertos
netstat -tuln | grep 3000          # Unix/Linux
netstat -ano | findstr :3000       # Windows

    Transición a Frameworks

    // HTTP Nativo: Control total, bajo overhead
import http from 'node:http';
const server = http.createServer(/* ... */);

// Express: Equilibrio entre simplicidad y features
import express from 'express';
const app = express();

// Fastify: Moderno, muy rápido
import fastify from 'fastify';
const app = fastify();

// NestJS: Enterprise, TypeScript, escalable
import { NestFactory } from '@nestjs/core';
const app = await NestFactory.create(AppModule);

// Cuándo elegir cada uno:
// - Aprender: HTTP nativo
// - MVP rápido: Express
// - Performance crítica: Fastify
// - Proyecto grande: NestJS
